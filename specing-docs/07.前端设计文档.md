```typescript
router.beforeEach((to, from, next) => {
  const authStore = useAuthStore()
  // 未登录 → 跳转登录页
  if (!to.meta.public && !authStore.isLoggedIn) {
    next('/login')
    return
  }
  // 角色权限检查
  const requiredRoles = to.meta.roles as string[] | undefined

  }
  next()
})
```

### 四、Axios 封装

#### 4.1 request.ts

```typescript
import axios from 'axios'
import { ElMessage } from 'element-plus'

const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE || '/api/v1',
  timeout: 30000,
})

// 请求拦截器：自动附加 Token
request.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器：统一错误处理
request.interceptors.response.use(
  (response) => {
    const { code, message, data } = response.data
    if (code !== 0) {
      ElMessage.error(message)
      if (code === 4001) {
        localStorage.removeItem('access_token')
        localStorage.removeItem('role')
        localStorage.removeItem('city')
        window.location.href = '/#/login'
      }
      if (code === 4003) {
        ElMessage.warning('无权限执行此操作')
      }
      return Promise.reject(new Error(message))
    }
    return data
  },
  (error) => {
    ElMessage.error('网络错误')
    return Promise.reject(error)
  }
)
```

### 五、状态管理（Pinia）

#### 5.1 auth store

```typescript
// stores/auth.ts
export type Role = 'super_admin' | 'city_admin' | 'viewer'

export const useAuthStore = defineStore('auth', () => {
  const token = ref(localStorage.getItem('access_token') || '')
  const adminInfo = ref<AdminInfo | null>(null)
  const role = ref<Role>(localStorage.getItem('role') as Role || 'viewer')
  const city = ref<string | null>(localStorage.getItem('city') || null)

  const isLoggedIn = computed(() => !!token.value)
  const isSuperAdmin = computed(() => role.value === 'super_admin')
  const isCityAdmin = computed(() => role.value === 'city_admin')
  const canOperate = computed(() => role.value !== 'viewer')  // 可操作/确认/修正
  const canExport = computed(() => role.value !== 'viewer')    // 可导出
  const canDelete = computed(() => role.value === 'super_admin') // 可删除
  const canManageAdmins = computed(() => role.value === 'super_admin') // 可管理账号

  // 地市操作权限：city_admin 只能操作自己地市的项目
  function canOperateCity(projectCity: string | null): boolean {
    if (role.value === 'super_admin') return true
    if (role.value === 'city_admin') return projectCity === city.value
    return false
  }

  async function login(username: string, password: string) {
    const res = await authApi.login(username, password)
    token.value = res.access_token
    adminInfo.value = res.admin
    role.value = res.admin.role
    city.value = res.admin.city ?? null
    // 持久化到 localStorage，刷新后恢复
    localStorage.setItem('access_token', res.access_token)
    localStorage.setItem('role', res.admin.role)
    localStorage.setItem('city', res.admin.city ?? '')
  }

  function logout() {
    token.value = ''
    adminInfo.value = null
    role.value = 'viewer'
    city.value = null
    localStorage.removeItem('access_token')
    localStorage.removeItem('role')
    localStorage.removeItem('city')
  }

  return {
    token, adminInfo, role, city,
    isLoggedIn, isSuperAdmin, isCityAdmin, canOperate, canExport, canDelete, canManageAdmins,
    canOperateCity,
    login, logout,
  }
})
```

**表单校验规则**：

- `username`：必填，3~50 字符，字母数字下划线

- `role`：必填，下拉选择
- `city`：role 为 `city_admin` 时必填，下拉选择（数据源 `GET /api/v1/cities`）；其他角色时隐藏/置空
- 不可删除自己的账号（前端做按钮禁用提示）
